kokoro==0.9.4

# NAMED EXPLICITLY ALTHOUGH KOKORO PULLS ALL THREE, because this is the chain
# that lets Kokoro pronounce a word outside its dictionary, and it fails
# SILENTLY: KPipeline builds its G2P with `unk=''`, tries to construct an
# espeak fallback, and catches and logs if it cannot. Without the fallback,
# names, paths and product names are dropped from speech with no error —
# "/home/wyatt/Projects/ixa" is spoken as "home Projects" and Ixa cannot say
# her own name. There is no system espeak-ng on the dev box; the library ships
# inside the venv in espeakng_loader. main.py probes this at startup and warns
# loudly, which is the other half of the same guard.
misaki==0.9.4
espeakng-loader==0.2.4
phonemizer-fork==3.3.2
